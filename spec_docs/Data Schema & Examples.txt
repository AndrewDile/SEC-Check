All examples:
/incidents:

Input OK:
{
  "Incident_ID": "102356304331",
  "Method": "NEW",
  "Timestamp": "2026-09-25T20:00:00Z",
  "Description": "Critical API gateway timeout linked to third-party auth provider.",
  "Materiality": "MATERIAL",
  "Severity": "HIGH",
  "Urgency": "URGENT"
}


Output 200 OK:
{
  "Incident_ID": "102356304331",
  "Status": "Created",
  "Description": "Critical API gateway timeout linked to third-party auth provider.",
  "Materiality": "MATERIAL",
  "Severity": "HIGH",
  "Urgency": "URGENT"
}

Input Bad: (After inputting same ID)
{
  "Incident_ID": "102356304331",
  "Method": "NEW",
  "Timestamp": "2026-09-25T20:00:00Z",
  "Description": "Critical API gateway timeout linked to third-party auth provider.",
  "Materiality": "MATERIAL",
  "Severity": "HIGH",
  "Urgency": "URGENT"
}

Output 400 Bad Request:
{
  "Error": "UUID_EXISTS",
  "Message": "The provided Incident_ID already exists in the system."
}
